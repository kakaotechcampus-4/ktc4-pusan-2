#
#   python -m venv .venv
#   ./.venv/Scripts/python.exe -m pip install -r requirements.txt
#   ./.venv/Scripts/python.exe -m pip install -e .
#
# This is the DEFAULT install and it is torch-free: the shipped mediapipe_geom
# backbone is pure geometry and needs neither torch nor a checkpoint. The 38
# tests that exercise the torch backends skip themselves. For those, see
# requirements-torch.txt.

# Korean on-screen text for pipeline_demo / collect / label_review. cv2.putText
# cannot draw Hangul, so these tools render it through Pillow plus a Windows CJK
# font (malgun.ttf / malgunbd.ttf / gulim.ttc) and fall back to English without
# it -- silently, which is why it must be declared rather than left implicit.
Pillow>=10.0

# matplotlib pulls kiwisolver. The 1.5.1 cp311 win_amd64 wheel fails to load on
# Windows 11 with "DLL load failed while importing _cext" (side-by-side
# configuration error) even with a current MSVC 2015-2022 x64 redistributable
# installed. Because mediapipe imports matplotlib transitively
# (tasks/python/vision/drawing_utils.py), that break takes down `import
# mediapipe` too and with it every preprocess code path. 1.4.9 loads correctly.
kiwisolver>=1.4,<1.5
